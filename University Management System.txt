#include <iostream>
#include <vector>
#include <fstream>
using namespace std;

struct BasicInfo {
    int id;
    char name[30];
    float marks;
};

class Student {
private:
    int id;
    string name;
    double cgpa;

    static int totalStudents;   // Static member
    const int createdNo;        // Const member

    friend void updateCGPA(Student &s, double newCGPA);   // Friend Function

public:
    // Default Constructor
    Student() : id(0), name("Unknown"), cgpa(0.0), createdNo(++totalStudents) {}

    // Parameterized Constructor
    Student(int i, string n, double c) : id(i), name(n), cgpa(c), createdNo(++totalStudents) {}

    // Destructor
    ~Student() {}

    // Function Overloading
    void setData(int i) { id = i; }
    void setData(int i, string n) { id = i; name = n; }
    void setData(int i, string n, double c) { id = i; name = n; cgpa = c; }

    // Operator Overloading → Compare CGPA
    bool operator>(const Student &other) {
        return this->cgpa > other.cgpa;
    }

    void display() const {
        cout << "ID: " << id << " | Name: " << name << " | CGPA: " << cgpa << endl;
    }

    string toFile() const {
        return to_string(id) + " " + name + " " + to_string(cgpa);
    }
};

int Student::totalStudents = 0;


/* ---------- FRIEND FUNCTION (Topic 6) ---------- */
void updateCGPA(Student &s, double newCGPA) {
    s.cgpa = newCGPA;
}

class Person {
public:
    string name;

    Person(string n = "XYZ") : name(n) {}

    virtual void show() {
        cout << "Person Name: " << name << endl;
    }
};

class Faculty : public Person {
public:
    string dept;

    Faculty(string n, string d) : Person(n), dept(d) {}

    void show() override {
        cout << "Faculty: " << name << " | Dept: " << dept << endl;
    }
};

template <typename T>
T add(T a, T b) {
    return a + b;
}

int main() {
    vector<Student> students;
    int choice;

    while (true) {
        cout << "\n===============================";
        cout << "\n   UNIVERSITY MANAGEMENT SYSTEM";
        cout << "\n===============================";
        cout << "\n1. Add Student's";
        cout << "\n2. Display All Students";
        cout << "\n3. Update CGPA ";
        cout << "\n4. Compare Students";
        cout << "\n5. Faculty Detail";
        cout << "\n6. Save Student's to File";
        cout << "\n7. Display Fees";
        cout << "\n8. Exit";
        cout << "\nEnter choice: ";
        cin >> choice;

        try {
            if (cin.fail()) throw "Invalid input!";

            if (choice == 1) {
                int id; string name; double cgpa;
                cout << "Enter ID: "; cin >> id;
                cout << "Enter Name: "; cin >> name;
                cout << "Enter CGPA: "; cin >> cgpa;

                Student s(id, name, cgpa);
                students.push_back(s);

                cout << "Student Added Successfully!\n";
            }

            else if (choice == 2) {
                cout << "\n--- STUDENT LIST ---\n";
                for (auto &s : students) s.display();
            }

            else if (choice == 3) {
                int id;
                double newCgpa;
                cout << "Enter index to update (0-based): ";
                cin >> id;
                cout << "Enter new CGPA: ";
                cin >> newCgpa;

                if (id < 0 || id >= students.size()) throw "Invalid student index!";
                
                updateCGPA(students[id], newCgpa);
                cout << "CGPA Updated!\n";
            }

            else if (choice == 4) {
                if (students.size() < 2) {
                    cout << "Need at least 2 students.\n";
                } else {
                    if (students[0] > students[1])
                        cout << students[0].toFile() << " has higher CGPA.\n";
                    else
                        cout << students[1].toFile() << " has higher CGPA.\n";
                }
            }

            else if (choice == 5) {
                Faculty f("Mohnish Sir", "AI&DS");
                f.show();
            }

            else if (choice == 6) {
                ofstream fout("university_students.txt");
                for (auto &s : students) fout << s.toFile() << endl;
                fout.close();

                cout << "Data Saved to university_students.txt\n";
            }

            else if (choice == 7) {
                cout << "Total Fees: " << add<int>(10000, 8000) << endl;
            }

            else if (choice == 8) {
                cout << "Exiting...\n";
                break;
            }

            else {
                throw "Invalid menu option!";
            }

        } catch (const char* msg) {
            cout << "Error: " << msg << endl;
            cin.clear();
            cin.ignore(1000, '\n');
        }
    }

    return 0;
}
